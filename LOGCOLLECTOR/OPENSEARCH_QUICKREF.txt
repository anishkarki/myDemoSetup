📊 WHAT EACH QUERY MATCHES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

FAILOVER QUERY (opensearch_failover_query.json):
  ✓ "failover" keyword
  ✓ "promoted" or "demoted" events
  ✓ "leader election" events
  ✓ "acquired lock" / "released lock"
  ✓ "promoted self to leader"
  ✓ "starting as a secondary"
  ✓ Pattern matching: failover variations

SWITCHOVER QUERY (opensearch_switchover_query.json):
  ✓ "switchover" keyword
  ✓ "manual failover" / "planned switchover"
  ✓ "received switchover request"
  ✓ "switchover scheduled"
  ✓ "demoting self"
  ✓ Pattern matching: switchover variations

ALL EVENTS QUERY (opensearch_all_events_query.json):
  ✓ Everything from failover + switchover queries
  ✓ "replication state changed"
  ✓ "timeline changed"
  ✓ "cluster topology changed"
  ✓ Includes HAProxy and etcd logs
  ✓ Aggregations by source, time, and severity

🎯 COMMON USE CASES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Get just the messages
curl -X GET "http://localhost:19200/patroni-logs-*/_search" \
  -H 'Content-Type: application/json' \
  -d @opensearch_failover_query.json | \
  jq -r '.hits.hits[]._source.message'

# Export to CSV
curl -X GET "http://localhost:19200/patroni-logs-*/_search" \
  -H 'Content-Type: application/json' \
  -d @opensearch_failover_query.json | \
  jq -r '.hits.hits[]._source | [.["@timestamp"],.source,.level,.message] | @csv' \
  > failover_logs.csv

# Count by source
curl -X GET "http://localhost:19200/patroni-logs-*/_search" \
  -H 'Content-Type: application/json' \
  -d @opensearch_all_events_query.json | \
  jq '.aggregations.events_by_source.buckets'

# Get last hour only
jq '.query.bool.filter += [{"range":{"@timestamp":{"gte":"now-1h"}}}]' \
  opensearch_failover_query.json > temp_query.json && \
curl -X GET "http://localhost:19200/patroni-logs-*/_search" \
  -H 'Content-Type: application/json' -d @temp_query.json | jq '.'

📈 QUERY FIELDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Returned fields:
  • @timestamp       - Event timestamp
  • message / log    - Log message content
  • source           - patroni1, patroni2, postgres, etc.
  • host             - Hostname
  • container_name   - Docker container name
  • level / severity - INFO, WARNING, ERROR, etc.
  • cluster          - Cluster name
  • timeline         - PostgreSQL timeline number
  • role             - leader, replica
